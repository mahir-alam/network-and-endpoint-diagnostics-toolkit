! =============================================================
! AccessSwitch1 - Cisco Catalyst 2960 (Layer 2 access switch)
! Role: access-layer switching for PC1 (VLAN 10) and PC2 (VLAN 20),
!       trunked uplink to DistSwitch1.
!
! Note: the 2960 only supports 802.1Q trunking, so unlike the 3560
! there is no "switchport trunk encapsulation dot1q" line needed
! here - that command doesn't exist/apply on this platform. This
! is intentional, not an inconsistency with DistSwitch1's config.
! =============================================================

enable
configure terminal

hostname AccessSwitch1

service password-encryption
enable secret Cisco12345
no ip domain-lookup

vlan 10
 name STAFF
exit
vlan 20
 name GUEST
exit
vlan 99
 name MGMT
exit

! --- Uplink to DistSwitch1 ---
interface FastEthernet0/1
 description Uplink-to-DistSwitch1
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
 no shutdown
exit

! --- Staff access ports ---
interface range FastEthernet0/2 - 11
 description STAFF-Access-Ports
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 no shutdown
exit

! --- Guest access ports ---
interface range FastEthernet0/12 - 20
 description GUEST-Access-Ports
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown
exit

! --- Management SVI (Layer 2 switch, no routing) ---
interface Vlan99
 description Management
 ip address 192.168.99.2 255.255.255.0
 no shutdown
exit
ip default-gateway 192.168.99.1

! --- Local admin + remote management ---
username admin privilege 15 secret AdminPass123
line con 0
 logging synchronous
 exec-timeout 5 0
exit
line vty 0 4
 login local
 transport input ssh
 exec-timeout 5 0
exit

end
write memory

! --- Verification commands to run and capture as evidence ---
! show vlan brief
! show interfaces trunk
! show ip interface brief
! show mac address-table
